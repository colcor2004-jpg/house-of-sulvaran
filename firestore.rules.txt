// Cópialas en Firebase Console → Firestore Database → Reglas

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Catálogo: cada producto es un documento independiente (products/{id}).
    // Lectura pública, escritura solo para usuarios autenticados (admin).
    match /products/{productId} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // Contenido del sitio y noticias (documentos pequeños).
    match /siteData/{docId} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // Clientes registrados.
    match /users/{userId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.auth.uid == userId;
      allow update: if request.auth != null && request.auth.uid == userId;
    }
  }
}
